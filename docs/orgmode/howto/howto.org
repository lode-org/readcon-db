Same job in every language: open a corpus, ingest a =.con=, select, read a hash.

** Rust
:PROPERTIES:
:CUSTOM_ID: rust
:END:
#+begin_src rust
use readcon_db::{ConCorpus, Select};

let db = ConCorpus::open("/tmp/corpus")?;
db.append_trajectory_path(1, "run.con")?;
let keys = db.select(
    &Select::new()
        .trajectory(1)
        .require_symbol("Cu")
        .natoms_range(1, 10_000)
        .limit(100),
)?;
let h = db.frame_hash(keys[0])?;
let frame = db.get_frame(keys[0])?;
#+end_src

Exact match: =Select::new().exact_hash(h.to_bytes())=.

** Python
:PROPERTIES:
:CUSTOM_ID: python
:END:
#+begin_src python
import readcon_db

db = readcon_db.ConCorpus.open("/tmp/corpus")
db.append_trajectory_path(1, "run.con")
sel = readcon_db.Select().require_symbol("Cu")
keys = db.select(sel)
h = db.frame_hash(keys[0])
text = db.get_frame_text(keys[0])
#+end_src

=maturin develop --features python= from a checkout. Module name is =readcon_db=.

** C / C++
:PROPERTIES:
:CUSTOM_ID: c-c
:END:
#+begin_src C
#include "readcon-db.h"

size_t id;
uint32_t n_frames;
rkrdb_open("/tmp/corpus", &id);
rkrdb_append_trajectory(id, 1, "run.con", &n_frames);
/* traj_id -1 matches every trajectory; limit 0 means no limit */
rkrdb_select_basic(id, -1, "Cu", 0, UINT32_MAX, 0);
for (int i = 0; i < rkrdb_result_count(id); i++) {
    uint64_t traj;
    uint32_t frame;
    rkrdb_result_key(id, i, &traj, &frame);
}
rkrdb_close(id);
#+end_src

Each call returns =RKRDB_OK= on success, and =rkrdb_last_error= copies the message of the last failure into a caller buffer.
C++ RAII: =readcon_db::Corpus=. MPI pack/bcast takes the *caller* communicator (=readcon-db-mpi.h=). The library does not call =MPI_Init=. Details: @@rst::doc:`/reference/c`@@.
